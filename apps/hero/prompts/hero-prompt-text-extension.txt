  <script>
    const heroPreview = document.getElementById("heroPreview");
    const heroH1 = document.getElementById("heroH1");
    const heroH2 = document.getElementById("heroH2");
    const darkAppearance = document.getElementById("darkAppearance");
    const centreText = document.getElementById("centreText");
    const minimumHeight = document.getElementById("minimumHeight");
    const showSubheading = document.getElementById("showSubheading");
    const copyButton = document.getElementById("copyButton");
    const copyStatus = document.getElementById("copyStatus");
    const fallbackPanel = document.getElementById("fallbackPanel");
    const generatedHTML = document.getElementById("generatedHTML");
    const heroStyles = document.getElementById("hero-styles");

    const settings = {
      dark: darkAppearance.checked,
      centre: centreText.checked,
      minimumHeight: minimumHeight.checked,
      showSubheading: showSubheading.checked
    };

    function updateHero() {
      settings.dark = darkAppearance.checked;
      settings.centre = centreText.checked;
      settings.minimumHeight = minimumHeight.checked;
      settings.showSubheading = showSubheading.checked;

      heroPreview.classList.toggle("hero-light", !settings.dark);
      heroPreview.classList.toggle("hero-dark", settings.dark);
      heroPreview.classList.toggle("hero-centre", settings.centre);
      heroPreview.classList.toggle("hero-min-height", settings.minimumHeight);

      heroH2.hidden = !settings.showSubheading;

      updateFallbackHTML();
    }

    function insertPlainText(event) {
      event.preventDefault();

      const plainText = event.clipboardData.getData("text/plain");
      const selection = window.getSelection();

      if (!selection || selection.rangeCount === 0) {
        return;
      }

      const range = selection.getRangeAt(0);
      range.deleteContents();

      const textNode = document.createTextNode(plainText);
      range.insertNode(textNode);

      range.setStartAfter(textNode);
      range.collapse(true);

      selection.removeAllRanges();
      selection.addRange(range);
    }

    function escapeHTML(text) {
      return text
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#39;");
    }

    function generateHTML() {
      const headingText = escapeHTML(heroH1.textContent);
      const subheadingText = escapeHTML(heroH2.textContent);

      const heroClasses = [
        "hero",
        settings.dark ? "hero-dark" : "hero-light",
        settings.centre ? "hero-centre" : "",
        settings.minimumHeight ? "hero-min-height" : ""
      ].filter(Boolean).join(" ");

      let subheadingHTML = "";

      if (settings.showSubheading) {
        subheadingHTML = `
          <h2>${subheadingText}</h2>
        `;
      }

      const exportedHeroCSS = heroStyles.textContent;

      const exportedDocument = `<!DOCTYPE html>
<html lang="en">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>My Hero Section</title>
  <meta name="description" content="Short, unique description of web page. Often used as the snippet shown in search results. About 120 characters is ideal.">
  <script src="https://cdn.jsdelivr.net/npm/@tailwindcss/browser@4"><\/script>
  <style type="text/tailwindcss">
${exportedHeroCSS}
  </style>
</head>
<body>
  <header class="${heroClasses}">
    <div class="hero-content">
      <h1>${headingText}</h1>
      ${subheadingHTML}
    </div>
  </header>
</body>
</html>`;

      return exportedDocument;
    }

    function updateFallbackHTML() {
      generatedHTML.value = generateHTML();
    }

    async function copyHTML() {
      const htmlToCopy = generateHTML();

      copyStatus.textContent = "";

      try {
        if (!navigator.clipboard || !navigator.clipboard.writeText) {
          throw new Error("Clipboard API unavailable");
        }

        await navigator.clipboard.writeText(htmlToCopy);

        copyStatus.textContent = "HTML copied—paste it into VS Code.";
        fallbackPanel.hidden = true;

      } catch (error) {
        generatedHTML.value = htmlToCopy;
        fallbackPanel.hidden = false;

        copyStatus.textContent =
          "Automatic copying is unavailable. Use the labelled textarea to copy your HTML manually.";

        fallbackPanel.scrollIntoView({
          behavior: "smooth",
          block: "nearest"
        });
      }
    }

    darkAppearance.addEventListener("change", updateHero);
    centreText.addEventListener("change", updateHero);
    minimumHeight.addEventListener("change", updateHero);
    showSubheading.addEventListener("change", updateHero);

    heroH1.addEventListener("input", updateFallbackHTML);
    heroH2.addEventListener("input", updateFallbackHTML);

    heroH1.addEventListener("paste", insertPlainText);
    heroH2.addEventListener("paste", insertPlainText);

    copyButton.addEventListener("click", copyHTML);

    updateHero();
  </script>
</body>
